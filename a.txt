# 1. 기존에 잘못 생성되어 깨진 파일 삭제
sudo rm -f /etc/apt/keyrings/docker.gpg

# 2. 올바른 공식 다운로드 주소(download.docker.com)로 다시 받기
curl -fsSL https://docker.com | sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg

# 3. Kali Linux 환경에 맞춰 데비안(bookworm) 저장소 강제 고정 등록
echo "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.gpg] https://docker.com bookworm stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

# 4. 저장소 갱신 및 도커 최신 버전 설치
sudo apt update
sudo apt install -y docker-ce docker-ce-cli containerd.io docker-compose-plugin
